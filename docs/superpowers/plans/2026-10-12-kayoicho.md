```ts
import type { LatLng, Point } from './types';

const EARTH_RADIUS_M = 6_371_000;
const METERS_PER_DEG_LAT = 111_320;

const toRad = (deg: number) => (deg * Math.PI) / 180;

/** 2点間の距離(メートル、ハバーサイン) */
export function distanceMeters(a: LatLng, b: LatLng): number {
  const dLat = toRad(b.lat - a.lat);
  const dLng = toRad(b.lng - a.lng);
  const h =
    Math.sin(dLat / 2) ** 2 + Math.cos(toRad(a.lat)) * Math.cos(toRad(b.lat)) * Math.sin(dLng / 2) ** 2;
  return 2 * EARTH_RADIUS_M * Math.asin(Math.min(1, Math.sqrt(h)));
}

export function centroid(points: readonly LatLng[]): LatLng {
  let lat = 0;
  let lng = 0;
  for (const p of points) {
    lat += p.lat;
    lng += p.lng;
  }
  return { lat: lat / points.length, lng: lng / points.length };
}

/** base から北へ north m、東へ east m ずらした点(短い距離用の近似) */
export function offsetMeters(base: LatLng, north: number, east: number): LatLng {
  return {
    lat: base.lat + north / METERS_PER_DEG_LAT,
    lng: base.lng + east / (METERS_PER_DEG_LAT * Math.cos(toRad(base.lat))),
  };
}

/** expo-location の LocationObject と同じ形(lib を expo に依存させないため構造で受ける) */
export type LocationLike = {
  coords: { latitude: number; longitude: number; accuracy: number | null };
  timestamp: number;
};

export function locationToPoint(loc: LocationLike): Point {
  return {
    lat: loc.coords.latitude,
    lng: loc.coords.longitude,
    accuracy: loc.coords.accuracy ?? null,
    t: loc.timestamp,
  };
}
```

- Produces: `detectStays(points: readonly Point[]): { stays: Stay[]; consumedUntil: number | null }`
  - `consumedUntil`: この時刻より前（`<`）の点は処理済みにしてよい。`null` なら確定できるものが無い（何も処理済みにしない）

ルール:
- 精度が `MAX_ACCURACY_M` 超の点は捨てる。時刻順に並べ直す。
- 先頭の点から塊を作り、次の点が塊の中心から `STAY_RADIUS_M` 以内なら塊に加えて中心を取り直す。
- 塊の外の点が来たら、到着＝塊の最初の点の時刻、出発＝その外の点の時刻。`出発−到着 >= STAY_MIN_MS` なら滞在。外の点から次の塊を始める。
- 最後の塊（外の点がまだ無い）は確定させず、`consumedUntil` = その塊の最初の点の時刻 として返す。

`src/lib/stay.ts`:
```ts
import { MAX_ACCURACY_M, STAY_MIN_MS, STAY_RADIUS_M } from './config';
import { centroid, distanceMeters } from './geo';
import type { Point, Stay } from './types';

export type DetectResult = {
  stays: Stay[];
  /** この時刻より前の点は処理済みにしてよい。null なら何も確定していない */
  consumedUntil: number | null;
};

/**
 * 座標の列から滞在を切り出す。
 * 出発時刻は「塊の外に出た最初の点」の時刻。止まっている間は位置更新が
 * 届かないので、次の点が来るまではそこにいたとみなす。
 */
export function detectStays(points: readonly Point[]): DetectResult {
  const good = points
    .filter((p) => p.accuracy === null || p.accuracy <= MAX_ACCURACY_M)
    .sort((a, b) => a.t - b.t);
  const stays: Stay[] = [];

  let i = 0;
  while (i < good.length) {
    const cluster: Point[] = [good[i]];
    let center = { lat: good[i].lat, lng: good[i].lng };
    let j = i + 1;
    while (j < good.length && distanceMeters(center, good[j]) <= STAY_RADIUS_M) {
      cluster.push(good[j]);
      center = centroid(cluster);
      j++;
    }
    if (j >= good.length) {
      // 最後の塊はまだ続いているかもしれないので次回に持ち越す
      return { stays, consumedUntil: good[i].t };
    }
    const arrivedAt = cluster[0].t;
    const departedAt = good[j].t;
    if (departedAt - arrivedAt >= STAY_MIN_MS) {
      stays.push({ lat: center.lat, lng: center.lng, arrivedAt, departedAt });
    }
    i = j;
  }
  return { stays, consumedUntil: null };
}
```

```ts
import { MERGE_GAP_MS, SAME_PLACE_RADIUS_M } from './config';
import { distanceMeters } from './geo';
import type { LatLng, Stay, Visit } from './types';

export type PlaceLocation = { id: number; lat: number; lng: number };

export type StayAction =
  | { type: 'extend'; visitId: number; departedAt: number }
  | { type: 'append'; placeId: number; arrivedAt: number; departedAt: number }
  | { type: 'newPlace'; lat: number; lng: number; arrivedAt: number; departedAt: number };

export function findPlace(places: readonly PlaceLocation[], at: LatLng): number | null {
  let best: number | null = null;
  let bestDistance = Infinity;
  for (const place of places) {
    const d = distanceMeters(place, at);
    if (d <= SAME_PLACE_RADIUS_M && d < bestDistance) {
      best = place.id;
      bestDistance = d;
    }
  }
  return best;
}

/**
 * 滞在1件をどう記録するか決める。
 * lastVisit は全建物を通して最も新しい訪問。同じ建物で短い途切れなら延長する
 * (建物の中でGPSが一瞬外れても2回に数えないため)。
 */
export function planStay(places: readonly PlaceLocation[], lastVisit: Visit | null, stay: Stay): StayAction {
  const placeId = findPlace(places, stay);
  if (placeId === null) {
    return { type: 'newPlace', lat: stay.lat, lng: stay.lng, arrivedAt: stay.arrivedAt, departedAt: stay.departedAt };
  }
  if (lastVisit && lastVisit.placeId === placeId && stay.arrivedAt - lastVisit.departedAt <= MERGE_GAP_MS) {
    return { type: 'extend', visitId: lastVisit.id, departedAt: Math.max(lastVisit.departedAt, stay.departedAt) };
  }
  return { type: 'append', placeId, arrivedAt: stay.arrivedAt, departedAt: stay.departedAt };
}
```

```ts
import type { Place, Visit } from './types';

export const UNNAMED = '名前のない場所';

export function displayName(place: Pick<Place, 'name'>): string {
  return place.name ?? UNNAMED;
}

export type PlaceStats = {
  placeId: number;
  name: string;
  lat: number;
  lng: number;
  count: number;
  totalMs: number;
  firstAt: number;
  lastAt: number;
};

export function summarize(places: readonly Place[], visits: readonly Visit[]): PlaceStats[] {
  const byPlace = new Map<number, PlaceStats>();
  const placeById = new Map(places.map((p) => [p.id, p]));
  for (const v of visits) {
    const place = placeById.get(v.placeId);
    if (!place) continue;
    const s = byPlace.get(v.placeId);
    if (s) {
      s.count += 1;
      s.totalMs += v.departedAt - v.arrivedAt;
      s.firstAt = Math.min(s.firstAt, v.arrivedAt);
      s.lastAt = Math.max(s.lastAt, v.arrivedAt);
    } else {
      byPlace.set(v.placeId, {
        placeId: place.id,
        name: displayName(place),
        lat: place.lat,
        lng: place.lng,
        count: 1,
        totalMs: v.departedAt - v.arrivedAt,
        firstAt: v.arrivedAt,
        lastAt: v.arrivedAt,
      });
    }
  }
  return [...byPlace.values()].sort((a, b) => a.placeId - b.placeId);
}

export type RankBy = 'count' | 'duration';

export function rank(stats: readonly PlaceStats[], by: RankBy): PlaceStats[] {
  const key = (s: PlaceStats) => (by === 'count' ? s.count : s.totalMs);
  return [...stats].sort((a, b) => key(b) - key(a) || b.lastAt - a.lastAt);
}

export type HistoryItem = { visitId: number; placeId: number; name: string; arrivedAt: number; departedAt: number };
export type DaySection = { dayKey: string; dayStart: number; items: HistoryItem[] };

const pad = (n: number) => String(n).padStart(2, '0');

export function groupByDay(places: readonly Place[], visits: readonly Visit[]): DaySection[] {
  const placeById = new Map(places.map((p) => [p.id, p]));
  const sorted = [...visits].sort((a, b) => b.arrivedAt - a.arrivedAt);
  const sections: DaySection[] = [];
  for (const v of sorted) {
    const d = new Date(v.arrivedAt);
    const dayKey = `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
    let section = sections[sections.length - 1];
    if (!section || section.dayKey !== dayKey) {
      section = { dayKey, dayStart: new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime(), items: [] };
      sections.push(section);
    }
    const place = placeById.get(v.placeId);
    section.items.push({
      visitId: v.id,
      placeId: v.placeId,
      name: place ? displayName(place) : UNNAMED,
      arrivedAt: v.arrivedAt,
      departedAt: v.departedAt,
    });
  }
  return sections;
}
```

```ts
const MIN = 60_000;
const WEEKDAYS = ['日', '月', '火', '水', '木', '金', '土'];

export function formatDuration(ms: number): string {
  const minutes = Math.floor(ms / MIN);
  if (minutes < 1) return '1分未満';
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  if (h === 0) return `${m}分`;
  return m === 0 ? `${h}時間` : `${h}時間${m}分`;
}

export function formatClock(t: number): string {
  const d = new Date(t);
  return `${d.getHours()}:${String(d.getMinutes()).padStart(2, '0')}`;
}

export function formatDate(t: number): string {
  const d = new Date(t);
  return `${d.getFullYear()}/${d.getMonth() + 1}/${d.getDate()}`;
}

function startOfDay(t: number): number {
  const d = new Date(t);
  return new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
}

export function formatDayTitle(dayStart: number, now: number): string {
  const today = startOfDay(now);
  const day = startOfDay(dayStart);
  if (day === today) return '今日';
  const yesterday = new Date(today);
  yesterday.setDate(yesterday.getDate() - 1);
  if (day === yesterday.getTime()) return '昨日';
  const d = new Date(day);
  const md = `${d.getMonth() + 1}月${d.getDate()}日(${WEEKDAYS[d.getDay()]})`;
  return d.getFullYear() === new Date(now).getFullYear() ? md : `${d.getFullYear()}年${md}`;
}
```

```ts
import { NAME_SEARCH_RADIUS_M } from './config';
import { distanceMeters } from './geo';
import type { LatLng } from './types';

export function buildOverpassQuery(at: LatLng, radius: number = NAME_SEARCH_RADIUS_M): string {
  return `[out:json][timeout:10];nwr(around:${radius},${at.lat},${at.lng})[name];out center tags;`;
}

const PLACE_KEYS = ['building', 'shop', 'amenity', 'tourism', 'office', 'leisure', 'healthcare', 'craft', 'public_transport'];

function isPlaceLike(tags: Record<string, unknown>): boolean {
  if (tags.railway === 'station') return true;
  return PLACE_KEYS.some((key) => key in tags);
}

function positionOf(el: Record<string, unknown>): LatLng | null {
  if (typeof el.lat === 'number' && typeof el.lon === 'number') return { lat: el.lat, lng: el.lon };
  const center = el.center as Record<string, unknown> | undefined;
  if (center && typeof center.lat === 'number' && typeof center.lon === 'number') {
    return { lat: center.lat, lng: center.lon };
  }
  return null;
}

/** Overpass の応答から、最も近い名前付きの建物・お店の名前を選ぶ */
export function pickName(json: unknown, at: LatLng): string | null {
  if (typeof json !== 'object' || json === null) return null;
  const elements = (json as { elements?: unknown }).elements;
  if (!Array.isArray(elements)) return null;

  let best: string | null = null;
  let bestDistance = Infinity;
  for (const raw of elements) {
    if (typeof raw !== 'object' || raw === null) continue;
    const el = raw as Record<string, unknown>;
    const tags = el.tags as Record<string, unknown> | undefined;
    if (!tags || !isPlaceLike(tags)) continue;
    const name = typeof tags['name:ja'] === 'string' ? tags['name:ja'] : tags.name;
    if (typeof name !== 'string' || name.trim() === '') continue;
    const pos = positionOf(el);
    if (!pos) continue;
    const d = distanceMeters(at, pos);
    if (d < bestDistance) {
      best = name.trim();
      bestDistance = d;
    }
  }
  return best;
}
```

```ts
import * as SQLite from 'expo-sqlite';
import type { NameSource, Place, Point, Visit } from '../lib/types';

let dbPromise: Promise<SQLite.SQLiteDatabase> | null = null;

export function getDb(): Promise<SQLite.SQLiteDatabase> {
  if (!dbPromise) dbPromise = open();
  return dbPromise;
}

async function open(): Promise<SQLite.SQLiteDatabase> {
  const db = await SQLite.openDatabaseAsync('kayoicho.db');
  await db.execAsync(`
    PRAGMA journal_mode = WAL;
    CREATE TABLE IF NOT EXISTS points (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      lat REAL NOT NULL,
      lng REAL NOT NULL,
      accuracy REAL,
      t INTEGER NOT NULL,
      processed INTEGER NOT NULL DEFAULT 0
    );
    CREATE INDEX IF NOT EXISTS points_processed_t ON points (processed, t);
    CREATE TABLE IF NOT EXISTS places (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT,
      name_source TEXT NOT NULL DEFAULT 'none',
      lat REAL NOT NULL,
      lng REAL NOT NULL,
      created_at INTEGER NOT NULL,
      naming_pending INTEGER NOT NULL DEFAULT 1
    );
    CREATE TABLE IF NOT EXISTS visits (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      place_id INTEGER NOT NULL REFERENCES places (id),
      arrived_at INTEGER NOT NULL,
      departed_at INTEGER NOT NULL
    );
    CREATE INDEX IF NOT EXISTS visits_departed ON visits (departed_at);
  `);
  return db;
}

export async function transaction(fn: () => Promise<void>): Promise<void> {
  const db = await getDb();
  await db.withTransactionAsync(fn);
}

type PointRow = { lat: number; lng: number; accuracy: number | null; t: number };
type PlaceRow = {
  id: number;
  name: string | null;
  name_source: string;
  lat: number;
  lng: number;
  created_at: number;
  naming_pending: number;
};
type VisitRow = { id: number; place_id: number; arrived_at: number; departed_at: number };

const toPlace = (r: PlaceRow): Place => ({
  id: r.id,
  name: r.name,
  nameSource: (r.name_source === 'osm' || r.name_source === 'user' ? r.name_source : 'none') as NameSource,
  lat: r.lat,
  lng: r.lng,
  createdAt: r.created_at,
  namingPending: r.naming_pending === 1,
});
const toVisit = (r: VisitRow): Visit => ({
  id: r.id,
  placeId: r.place_id,
  arrivedAt: r.arrived_at,
  departedAt: r.departed_at,
});

export async function insertPoints(points: readonly Point[]): Promise<void> {
  if (points.length === 0) return;
  const db = await getDb();
  await db.withTransactionAsync(async () => {
    for (const p of points) {
      await db.runAsync('INSERT INTO points (lat, lng, accuracy, t) VALUES (?, ?, ?, ?)', p.lat, p.lng, p.accuracy, p.t);
    }
  });
}

export async function getUnprocessedPoints(): Promise<Point[]> {
  const db = await getDb();
  return db.getAllAsync<PointRow>('SELECT lat, lng, accuracy, t FROM points WHERE processed = 0 ORDER BY t');
}

export async function markProcessedBefore(t: number): Promise<void> {
  const db = await getDb();
  await db.runAsync('UPDATE points SET processed = 1 WHERE processed = 0 AND t < ?', t);
}

export async function prunePoints(before: number): Promise<void> {
  const db = await getDb();
  await db.runAsync('DELETE FROM points WHERE processed = 1 AND t < ?', before);
}

export async function getPlaces(): Promise<Place[]> {
  const db = await getDb();
  return (await db.getAllAsync<PlaceRow>('SELECT * FROM places ORDER BY id')).map(toPlace);
}

export async function getVisits(): Promise<Visit[]> {
  const db = await getDb();
  return (await db.getAllAsync<VisitRow>('SELECT * FROM visits ORDER BY arrived_at')).map(toVisit);
}

export async function getLastVisit(): Promise<Visit | null> {
  const db = await getDb();
  const row = await db.getFirstAsync<VisitRow>('SELECT * FROM visits ORDER BY departed_at DESC LIMIT 1');
  return row ? toVisit(row) : null;
}

export async function insertPlace(lat: number, lng: number, createdAt: number): Promise<number> {
  const db = await getDb();
  const result = await db.runAsync('INSERT INTO places (lat, lng, created_at) VALUES (?, ?, ?)', lat, lng, createdAt);
  return result.lastInsertRowId;
}

export async function insertVisit(placeId: number, arrivedAt: number, departedAt: number): Promise<void> {
  const db = await getDb();
  await db.runAsync('INSERT INTO visits (place_id, arrived_at, departed_at) VALUES (?, ?, ?)', placeId, arrivedAt, departedAt);
}

export async function extendVisit(visitId: number, departedAt: number): Promise<void> {
  const db = await getDb();
  await db.runAsync('UPDATE visits SET departed_at = ? WHERE id = ?', departedAt, visitId);
}

export async function getPendingNaming(limit: number): Promise<Place[]> {
  const db = await getDb();
  const rows = await db.getAllAsync<PlaceRow>(
    "SELECT * FROM places WHERE naming_pending = 1 AND name_source != 'user' ORDER BY id LIMIT ?",
    limit,
  );
  return rows.map(toPlace);
}

/** 自動でつけた名前を保存する。ユーザーがつけた名前は上書きしない */
export async function setOsmName(placeId: number, name: string | null): Promise<void> {
  const db = await getDb();
  await db.runAsync(
    "UPDATE places SET name = ?, name_source = ?, naming_pending = 0 WHERE id = ? AND name_source != 'user'",
    name,
    name === null ? 'none' : 'osm',
    placeId,
  );
}

export async function renamePlace(placeId: number, name: string): Promise<void> {
  const db = await getDb();
  await db.runAsync("UPDATE places SET name = ?, name_source = 'user', naming_pending = 0 WHERE id = ?", name, placeId);
}
```

```ts
import { buildOverpassQuery, pickName } from '../lib/naming';
import type { LatLng } from '../lib/types';

const ENDPOINT = 'https://overpass-api.de/api/interpreter';
const TIMEOUT_MS = 15_000;

export type NameLookup = { ok: true; name: string | null } | { ok: false };

/** 失敗(通信・タイムアウト・HTTPエラー)は ok:false。名前が無かっただけなら ok:true, name:null */
export async function lookupName(at: LatLng): Promise<NameLookup> {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS);
  try {
    const res = await fetch(ENDPOINT, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'User-Agent': 'kayoicho/0.1' },
      body: `data=${encodeURIComponent(buildOverpassQuery(at))}`,
      signal: controller.signal,
    });
    if (!res.ok) return { ok: false };
    return { ok: true, name: pickName(await res.json(), at) };
  } catch {
    return { ok: false };
  } finally {
    clearTimeout(timer);
  }
}
```

```ts
import { POINT_RETENTION_MS } from '../lib/config';
import { planStay, type StayAction } from '../lib/match';
import { detectStays } from '../lib/stay';
import * as db from './db';
import { lookupName } from './overpass';

/** 1回の処理で名前を問い合わせる建物の数(Overpass に負荷をかけないため) */
const NAMING_PER_RUN = 3;

let running: Promise<void> | null = null;

/** 未処理の座標 → 滞在 → 建物・訪問 を反映し、名前の無い建物に名前をつける */
export function processPending(now: number = Date.now()): Promise<void> {
  if (!running) {
    running = run(now).finally(() => {
      running = null;
    });
  }
  return running;
}

async function apply(action: StayAction, now: number): Promise<void> {
  switch (action.type) {
    case 'extend':
      await db.extendVisit(action.visitId, action.departedAt);
      return;
    case 'append':
      await db.insertVisit(action.placeId, action.arrivedAt, action.departedAt);
      return;
    case 'newPlace': {
      const placeId = await db.insertPlace(action.lat, action.lng, now);
      await db.insertVisit(placeId, action.arrivedAt, action.departedAt);
      return;
    }
  }
}

async function run(now: number): Promise<void> {
  const points = await db.getUnprocessedPoints();
  const { stays, consumedUntil } = detectStays(points);

  // 訪問の記録と「処理済み」の印は一緒に確定させる(途中で落ちて二重に数えないため)
  await db.transaction(async () => {
    for (const stay of stays) {
      const [places, lastVisit] = await Promise.all([db.getPlaces(), db.getLastVisit()]);
      await apply(planStay(places, lastVisit, stay), now);
    }
    if (consumedUntil !== null) await db.markProcessedBefore(consumedUntil);
  });

  await db.prunePoints(now - POINT_RETENTION_MS);

  for (const place of await db.getPendingNaming(NAMING_PER_RUN)) {
    const result = await lookupName(place);
    // 失敗したら naming_pending のまま残し、次回また問い合わせる
    if (result.ok) await db.setOsmName(place.id, result.name);
  }
}
```

```ts
import * as Location from 'expo-location';
import * as TaskManager from 'expo-task-manager';
import { locationToPoint } from '../lib/geo';
import { insertPoints } from './db';
import { processPending } from './process';

export const TASK_NAME = 'kayoicho-location';

// アプリが裏で起こされたときにも登録されているよう、モジュール読み込み時に定義する
TaskManager.defineTask<{ locations: Location.LocationObject[] }>(TASK_NAME, async ({ data, error }) => {
  if (error || !data) return;
  await insertPoints(data.locations.map(locationToPoint));
  await processPending();
});

/**
 * notAsked: まだ一度も聞いていない / needsAlways: 「使用中のみ」で止まっている
 * denied: 拒否 / unavailable: 権限はあるが位置タスクを始められない(Expo Go など)
 */
export type TrackingState = 'on' | 'notAsked' | 'needsAlways' | 'denied' | 'unavailable';

export async function getTrackingState(): Promise<TrackingState> {
  const fg = await Location.getForegroundPermissionsAsync();
  if (fg.status === 'undetermined') return 'notAsked';
  if (fg.status !== 'granted') return 'denied';
  const bg = await Location.getBackgroundPermissionsAsync();
  if (bg.status !== 'granted') return 'needsAlways';
  return (await isStarted()) ? 'on' : 'unavailable';
}

async function isStarted(): Promise<boolean> {
  try {
    return await Location.hasStartedLocationUpdatesAsync(TASK_NAME);
  } catch {
    return false;
  }
}

/** 権限を求め(まだなら)、位置タスクを始める。結果の状態を返す */
export async function startTracking(): Promise<TrackingState> {
  const fg = await Location.requestForegroundPermissionsAsync();
  if (fg.status !== 'granted') return 'denied';
  const bg = await Location.requestBackgroundPermissionsAsync();
  if (bg.status !== 'granted') return 'needsAlways';
  if (await isStarted()) return 'on';
  try {
    await Location.startLocationUpdatesAsync(TASK_NAME, {
      accuracy: Location.Accuracy.Balanced,
      distanceInterval: 20,
      pausesUpdatesAutomatically: false,
      activityType: Location.ActivityType.Other,
      showsBackgroundLocationIndicator: false,
      foregroundService: {
        notificationTitle: 'かよい帳',
        notificationBody: '行った建物を記録しています',
      },
    });
  } catch {
    return 'unavailable';
  }
  return 'on';
}
```

```tsx
import React, { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import { AppState } from 'react-native';
import type { Place, Visit } from '../lib/types';
import { getPlaces, getVisits, renamePlace } from '../services/db';
import { processPending } from '../services/process';
import { getTrackingState, startTracking, type TrackingState } from '../services/tracker';

type DataValue = {
  places: Place[];
  visits: Visit[];
  tracking: TrackingState | null;
  loading: boolean;
  refresh: () => Promise<void>;
  enableTracking: () => Promise<void>;
  rename: (placeId: number, name: string) => Promise<void>;
};

const DataContext = createContext<DataValue | null>(null);

export function DataProvider({ children }: { children: React.ReactNode }) {
  const [places, setPlaces] = useState<Place[]>([]);
  const [visits, setVisits] = useState<Visit[]>([]);
  const [tracking, setTracking] = useState<TrackingState | null>(null);
  const [loading, setLoading] = useState(true);

  const load = useCallback(async () => {
    const [p, v] = await Promise.all([getPlaces(), getVisits()]);
    setPlaces(p);
    setVisits(v);
  }, []);

  const refresh = useCallback(async () => {
    try {
      await processPending();
    } finally {
      await load();
      setLoading(false);
    }
  }, [load]);

  const enableTracking = useCallback(async () => {
    setTracking(await startTracking());
  }, []);

  const rename = useCallback(
    async (placeId: number, name: string) => {
      await renamePlace(placeId, name);
      await load();
    },
    [load],
  );

  useEffect(() => {
    (async () => {
      const state = await getTrackingState();
      // 権限がすでにあるなら黙って記録を始める。まだ聞いていないなら帯で理由を説明してから聞く
      setTracking(state === 'unavailable' ? await startTracking() : state);
      await refresh();
    })();
    const sub = AppState.addEventListener('change', (next) => {
      if (next !== 'active') return;
      getTrackingState().then(setTracking);
      refresh();
    });
    return () => sub.remove();
  }, [refresh]);

  const value = useMemo(
    () => ({ places, visits, tracking, loading, refresh, enableTracking, rename }),
    [places, visits, tracking, loading, refresh, enableTracking, rename],
  );
  return <DataContext.Provider value={value}>{children}</DataContext.Provider>;
}

export function useData(): DataValue {
  const value = useContext(DataContext);
  if (!value) throw new Error('useData は DataProvider の中で使うこと');
  return value;
}
```

```ts
import type { PlaceStats } from './stats';

export type MapMarker = { id: number; lat: number; lng: number; radius: number; label: string };

/** 回数が多いほど大きく。倍になるごとに+4px、24pxで頭打ち */
export function markerRadius(count: number): number {
  return Math.min(24, 6 + 4 * Math.log2(Math.max(1, count)));
}

export function toMarkers(stats: readonly PlaceStats[]): MapMarker[] {
  return stats.map((s) => ({
    id: s.placeId,
    lat: s.lat,
    lng: s.lng,
    radius: markerRadius(s.count),
    label: `${s.name}・${s.count}回`,
  }));
}
```

```ts
/**
 * HTML の script タグの中に埋めても壊れない JSON。
 * `<` `>` を潰すことで `</script>` による脱出を、U+2028/U+2029 を潰すことで
 * JS の行区切り扱いを防ぐ。どれも \uXXXX なので読み戻すと元に戻る。
 */
export function safeJson(value: unknown): string {
  return JSON.stringify(value)
    .replace(/</g, '\\u003c')
    .replace(/>/g, '\\u003e')
    .replace(/ /g, '\\u2028')
    .replace(/ /g, '\\u2029');
}

/** マーカーが増えても作り直さないこと(WebView がリロードされる)。差し替えは window.kayoi.setMarkers */
export function buildMapHtml(): string {
  return `<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="utf-8" />
<meta name="viewport" content="width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no" />
<link rel="stylesheet" href="https://unpkg.com/leaflet@1.9.4/dist/leaflet.css" />
<style>
  html, body, #map { height: 100%; margin: 0; padding: 0; background: #EDEAE4; }
  .leaflet-container { background: #EDEAE4; }
  .leaflet-control-attribution { font-size: 9px; }
</style>
</head>
<body>
<div id="map"></div>
<script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>
<script>
(function () {
  function post(message) {
    if (window.ReactNativeWebView) window.ReactNativeWebView.postMessage(JSON.stringify(message));
  }
  if (typeof L === 'undefined') { post({ type: 'loadFailed' }); return; }

  try {
    var map = L.map('map', { zoomControl: false }).setView([35.681236, 139.767125], 12);
    map.attributionControl.setPrefix('');
    L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png', {
      maxZoom: 19,
      attribution: '&copy; OpenStreetMap'
    }).addTo(map);

    var layer = L.layerGroup().addTo(map);
    var fitted = false;

    window.kayoi = {
      setMarkers: function (markers) {
        layer.clearLayers();
        var bounds = [];
        for (var i = 0; i < markers.length; i++) {
          var m = markers[i];
          var label = document.createElement('div');
          label.textContent = m.label;
          L.circleMarker([m.lat, m.lng], {
            radius: m.radius,
            fillColor: '#2F6B5E',
            fillOpacity: 0.55,
            color: '#FFFFFF',
            weight: 2
          }).bindPopup(label).addTo(layer);
          bounds.push([m.lat, m.lng]);
        }
        if (!fitted && bounds.length > 0) {
          map.fitBounds(bounds, { maxZoom: 16, padding: [40, 40] });
          fitted = true;
        }
      }
    };

    // コンテナの高さが後から決まると Leaflet が古いサイズのまま描くので測り直す
    var el = document.getElementById('map');
    if (typeof ResizeObserver !== 'undefined') {
      new ResizeObserver(function () {
        try { map.invalidateSize(); } catch (e) {}
      }).observe(el);
    }
    setTimeout(function () { try { map.invalidateSize(); } catch (e) {} }, 100);

    post({ type: 'ready' });
  } catch (e) {
    post({ type: 'loadFailed' });
  }
})();
</script>
</body>
</html>`;
}
```

```tsx
import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { StyleSheet, Text, View } from 'react-native';
import { WebView, type WebViewMessageEvent } from 'react-native-webview';
import { buildMapHtml, safeJson } from '../lib/leafletHtml';
import type { MapMarker } from '../lib/markers';
import { COLORS, SPACING } from '../ui/theme';

/** Leaflet が CDN から読めないと判断するまでの待ち時間 */
const LOAD_TIMEOUT_MS = 10_000;

export function PlaceMap({ markers }: { markers: readonly MapMarker[] }) {
  const webRef = useRef<WebView>(null);
  const [status, setStatus] = useState<'loading' | 'ready' | 'failed'>('loading');
  const html = useMemo(() => buildMapHtml(), []);

  const push = useCallback(() => {
    webRef.current?.injectJavaScript(`window.kayoi && window.kayoi.setMarkers(${safeJson(markers)}); true;`);
  }, [markers]);

  useEffect(() => {
    if (status === 'ready') push();
  }, [status, push]);

  useEffect(() => {
    const timer = setTimeout(() => setStatus((s) => (s === 'loading' ? 'failed' : s)), LOAD_TIMEOUT_MS);
    return () => clearTimeout(timer);
  }, []);

  const onMessage = (event: WebViewMessageEvent) => {
    let type: unknown;
    try {
      type = (JSON.parse(event.nativeEvent.data) as { type?: unknown }).type;
    } catch {
      return;
    }
    if (type === 'ready') setStatus('ready');
    if (type === 'loadFailed') setStatus((s) => (s === 'ready' ? s : 'failed'));
  };

  return (
    <View style={styles.container}>
      <WebView
        ref={webRef}
        style={styles.web}
        originWhitelist={['*']}
        source={{ html }}
        onMessage={onMessage}
        onError={() => setStatus('failed')}
        javaScriptEnabled
        scrollEnabled={false}
        setSupportMultipleWindows={false}
      />
      {status === 'failed' && (
        <View style={styles.overlay}>
          <Text style={styles.overlayText}>地図を読み込めませんでした。通信できる場所でもう一度開いてください。</Text>
        </View>
      )}
    </View>
  );
}
```

````markdown
# かよい帳

GPS で自分が行った建物を自動で記録し、履歴と「何回行ったか」を残す iPhone / Android アプリ（Expo SDK 56）。

- 建物に **5 分以上**いると「訪問 1 回」として記録（半径 50m）
- 建物の名前は OpenStreetMap（Overpass API）から自動でつける。見つからなければ「名前のない場所」、自分で名前を変えられる
- 履歴 / ランキング（回数順・時間順）/ 地図 の 3 タブ
- データはこの端末の中だけ（クラウドなし）。外に出るのは名前を探すための座標問い合わせと地図タイルの取得だけ

## 動かし方

Expo Go の iOS 版はバックグラウンドで位置を取れないので、開発用ビルドで動かす。

```bash
pnpm install
pnpm ios        # 開発用ビルドを作ってシミュレータで起動（2回目以降は pnpm start でも可）
pnpm test       # 判定ロジックのテスト
```

シミュレータで位置を動かす:

```bash
xcrun simctl privacy booted grant location-always com.nokokoro.kayoicho
xcrun simctl location booted set 35.681236,139.767125
```

## 仕組み

`src/lib` に判断をすべて置き、React/Expo を import しない（テストはここだけに当てる）。

1. 位置タスク（`src/services/tracker.ts`）が座標を受け取り、端末 DB に追記
2. `processPending()` が未処理の座標から滞在を切り出し（`stay.ts`）、既知の建物か判定（`match.ts`）、訪問を記録
3. 新しい建物は Overpass で名前を探す（`naming.ts`）。失敗したら次回また探す

止まっている間は位置更新が届かないので、「次の点が来るまではそこにいた」とみなしている。

## 制約

- iOS は電池節約のため裏での位置更新を間引くことがあり、数分だけ寄った場所は取りこぼすことがある
- 建物の中でGPSが外れても 10 分以内に戻れば 1 回にまとめるが、それより長いと 2 回に数える
- 民家・小さな建物は OpenStreetMap に名前が無いことが多い
- 処理済みの生の座標は 30 日で消える（訪問の記録は残る）
````
